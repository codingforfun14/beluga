// omografia: rettangolo 1000x620 -> quattro angoli in pixel (tl, tr, br, bl)
function homography(p){
  const W=1000,H=620,[x0,y0]=p[0],[x1,y1]=p[1],[x2,y2]=p[2],[x3,y3]=p[3];
  const dx1=x1-x2,dx2=x3-x2,dy1=y1-y2,dy2=y3-y2,sx=x0-x1+x2-x3,sy=y0-y1+y2-y3;
  const den=dx1*dy2-dx2*dy1, g=(sx*dy2-dx2*sy)/den, h=(dx1*sy-sx*dy1)/den;
  const a=x1-x0+g*x1, b=x3-x0+h*x3, c=x0, d=y1-y0+g*y1, e=y3-y0+h*y3, f=y0;
  const m=[a/W,d/W,0,g/W, b/H,e/H,0,h/H, 0,0,1,0, c,f,0,1];
  return 'matrix3d('+m.map(v=>+v.toFixed(8)).join(',')+')';
}
